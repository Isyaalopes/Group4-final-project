FINALS Lab Exercise #2 — Exploring XMLHttpRequest

HTML & JavaScript Code:

<!DOCTYPE html> <html lang="en"> <head> <meta charset="UTF-8"> <meta name="viewport" content="width=device-width, initial-scale=1.0"> <title>Lab Exercise 2 — XMLHttpRequest</title> <link rel="stylesheet" href="style.css"> </head> <body> <h2>Exploring XMLHttpRequest</h2> <button id="loadData">Load Post Data</button> <div id="output"> Click the button above to load content... </div> <script> const loadBtn = document.getElementById('loadData'); const outputDiv = document.getElementById('output'); const apiEndpoint = 'https://jsonplaceholder.typicode.com/posts/1'; loadBtn.addEventListener('click', fetchPost); function fetchPost() { const xhr = new XMLHttpRequest(); xhr.onreadystatechange = function () { if (xhr.readyState === 4) { if (xhr.status === 200) { try { const postData = JSON.parse(xhr.responseText); outputDiv.innerHTML = ` <strong>Post Title:</strong><br> ${postData.title} <br><br> <strong>Post Body:</strong><br> ${postData.body} `; } catch (err) { outputDiv.innerHTML = `<span class="error-text">Failed to parse server response.</span>`; } } else { outputDiv.innerHTML = `<span class="error-text"> Request failed — Status ${xhr.status}: ${xhr.statusText} </span>`; } } }; xhr.open('GET', apiEndpoint, true); xhr.send(); } </script> </body> </html>

FINALS Lab Exercise #3 — Exploring the Fetch API

HTML & JavaScript Code:

<!DOCTYPE html> <html lang="en"> <head> <meta charset="UTF-8"> <meta name="viewport" content="width=device-width, initial-scale=1.0"> <title>Lab Exercise 3 — Fetch API</title> <link rel="stylesheet" href="style.css"> </head> <body> <h2>Exploring the Fetch API</h2> <button id="loadData">Load Post Data</button> <button id="createPost">Create New Post</button> <div id="output"> Click a button above to begin... </div> <script> const outputDiv = document.getElementById('output'); const loadBtn = document.getElementById('loadData'); const createBtn = document.getElementById('createPost'); const getLink = 'https://jsonplaceholder.typicode.com/posts/1'; const postLink = 'https://jsonplaceholder.typicode.com/posts'; loadBtn.addEventListener('click', () => { outputDiv.textContent = 'Loading post...'; fetch(getLink) .then(res => { if (!res.ok) throw new Error(`Status ${res.status}`); return res.json(); }) .then(data => { outputDiv.innerHTML = ` <strong>Post Title:</strong><br> ${data.title} <br><br> <strong>Content:</strong><br> ${data.body} `; }) .catch(err => { outputDiv.innerHTML = `<span class="error-text">Error: ${err.message}</span>`; }); }); createBtn.addEventListener('click', () => { outputDiv.textContent = 'Sending new post...'; const newPost = { title: "My New Post", body: "Learning Fetch API is fun!", userId: 1 }; fetch(postLink, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(newPost) }) .then(res => { if (!res.ok) throw new Error(`Status ${res.status}`); return res.json(); }) .then(data => { outputDiv.innerHTML = ` <span class="success-text">New Post Created!</span> <br><br> <strong>Post ID:</strong> ${data.id} `; }) .catch(err => { outputDiv.innerHTML = `<span class="error-text">Error: ${err.message}</span>`; }); }); </script> </body> </html>
